# Runtime and test dependencies for the eventstore exercise.
#
# Two stores, two reasons.
#
# MongoDB is the operational store: a server, reached over the network, holding the raw
# event stream. pymongo talks to it. DuckDB is the analytical store: a library, holding
# the rollups in one file, with no server to run and nothing to connect to.
#
# No numpy, no scipy, no pandas. Every statistic this service reports is in
# src/eventstore/perfmodel/ as a pure function over lists of floats, which is what lets
# the tests assert hand-computed constants instead of comparing against a snapshot that
# nobody can derive. A statistics dependency would move those numbers somewhere they
# cannot be reviewed.
fastapi>=0.110,<1
uvicorn[standard]>=0.27,<1
pydantic>=2.6,<3
pymongo>=4.6,<5
duckdb>=1.1,<2

# httpx is what fastapi.testclient drives the app through; it is a test dependency that
# the application itself never imports.
httpx>=0.27,<1
pytest>=8.0,<9
